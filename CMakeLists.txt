cmake_minimum_required(VERSION 3.16)
include($ENV{IDF_PATH}/tools/cmake/project.cmake)

project(FoloToy-AI-Passport)

# Keep narration outside the 3 MB application partition. Register the staged
# image with IDF so both flash_args and merge-bin contain exactly this payload.
set(DINO_AUDIO_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/assets/audio/dinobook40/audio.bin")
set(DINO_AUDIO_IMAGE "${CMAKE_BINARY_DIR}/dino_audio/audio.bin")
if(EXISTS "${DINO_AUDIO_SOURCE}")
    file(SIZE "${DINO_AUDIO_SOURCE}" DINO_AUDIO_SIZE)
    partition_table_get_partition_info(DINO_AUDIO_LIMIT "--partition-name dino_audio" "size")
    if(NOT DINO_AUDIO_LIMIT)
        message(FATAL_ERROR "The dino_audio partition is missing")
    endif()
    math(EXPR DINO_AUDIO_LIMIT_BYTES "${DINO_AUDIO_LIMIT}")
    if(DINO_AUDIO_SIZE LESS 1)
        message(FATAL_ERROR "DinoBook narration bank must not be empty")
    elseif(DINO_AUDIO_SIZE GREATER DINO_AUDIO_LIMIT_BYTES)
        message(FATAL_ERROR "DinoBook narration bank (${DINO_AUDIO_SIZE} bytes) exceeds its ${DINO_AUDIO_LIMIT_BYTES}-byte partition")
    endif()
    configure_file("${DINO_AUDIO_SOURCE}" "${DINO_AUDIO_IMAGE}" COPYONLY)
    esptool_py_flash_to_partition(flash dino_audio "${DINO_AUDIO_IMAGE}")
else()
    message(WARNING "Narration bank omitted: building public source profile without bundled audio. Add a compatible, redistributable narration bank locally to enable audio.")
endif()
